## 4. 取证方法（可复跑）

```bash
# 静默异常 / 裸 except / 超长函数：AST 统计（Python）
python - <<'PY'
import ast, pathlib, collections
files = ["UnipusAI_Helper.py", "time_brushing.py"] + list(map(str, pathlib.Path("unipus_helper").glob("*.py")))
silent, bare = collections.Counter(), collections.Counter()
for f in files:
    for n in ast.walk(ast.parse(pathlib.Path(f).read_text(encoding="utf-8"))):
        if isinstance(n, ast.ExceptHandler):
            kinds = {type(x).__name__ for x in n.body}
            if kinds <= {"Pass"} or kinds <= {"Continue"}:
                silent[f] += 1
            if n.type is None:
                bare[f] += 1
print("silent:", sum(silent.values()), dict(silent))
print("bare:", sum(bare.values()), dict(bare))
PY
```
